Async Programming & The Event Loop
JS is single-threaded — one task runs at a time. The event loop is how it handles async operations without blocking.

The JavaScript runtime model

JS has one call stack — it executes one thing at a time. Long tasks would block the UI. The solution: async operations (timers, network, I/O) are handed off to browser APIs. When done, their callbacks go into a task queue. The event loop constantly checks: if the call stack is empty, it takes the next callback from the queue and runs it. Microtasks (Promises) have a higher-priority queue that drains completely before the next macrotask.

event_loop.js 
// ── Execution order — the core mental model ───────────────────

console.log("1 — synchronous");

setTimeout(() => {
  console.log("4 — macrotask (setTimeout)");
}, 0);
// Even with 0ms delay, this goes to the task queue
// and runs AFTER synchronous code + microtasks

Promise.resolve().then(() => {
  console.log("3 — microtask (Promise .then)");
});
// Microtasks drain before any macrotask — higher priority

console.log("2 — synchronous");

// Output order:
// 1 — synchronous
// 2 — synchronous
// 3 — microtask (Promise .then)
// 4 — macrotask (setTimeout)

// ── Why this matters ──────────────────────────────────────────
// Microtasks = Promise .then/.catch, queueMicrotask()
// Macrotasks = setTimeout, setInterval, I/O, UI events
//
// All microtasks run to completion before the next macrotask.
// So if you queue many microtasks in a loop, they all run first.

// ── Never block the call stack ────────────────────────────────
// ❌ This freezes the browser for 3 seconds
function blockingWork() {
  const end = Date.now() + 3000;
  while (Date.now() < end) {}  // spin-wait — BAD
}

// ✅ Break work into async chunks instead
async function chunkedWork(items) {
  for (const item of items) {
    processItem(item);
    await new Promise(resolve => setTimeout(resolve, 0));
    // ↑ yield to event loop each iteration — keeps UI responsive
  }
}
execution order
▶1 — synchronous
▶2 — synchronous
▶3 — microtask (Promise .then)
▶4 — macrotask (setTimeout)
// 01
setTimeout & setInterval
Browser timer APIs — schedule code to run once (setTimeout) or repeatedly (setInterval).
timers.js 
// ── setTimeout — run once after delay ────────────────────────
const id1 = setTimeout(() => {
  console.log("Runs after ~1000ms");
}, 1000);

// Cancel before it fires
clearTimeout(id1);  // id1 is now cancelled — won't run

// Pass arguments to callback
setTimeout((name, greeting) => {
  console.log(`${greeting}, ${name}!`);
}, 500, "Alice", "Hello");  // extra args after delay

// Minimum delay is not guaranteed — it's "at least N ms"
// Actual firing depends on call stack being empty

// ── setInterval — run repeatedly ─────────────────────────────
let count = 0;
const id2 = setInterval(() => {
  count++;
  console.log(`tick ${count}`);
  if (count >= 3) clearInterval(id2);  // stop after 3 ticks
}, 1000);
// tick 1 → tick 2 → tick 3 → stops

// ── setInterval problem: drift ────────────────────────────────
// If callback takes longer than interval, calls overlap/drift
// Prefer recursive setTimeout for precise timing:
function preciseTimer() {
  console.log("tick");
  setTimeout(preciseTimer, 1000);  // next tick only AFTER this one finishes
}

// ── Debounce — delay until user stops ────────────────────────
function debounce(fn, delay) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}
const search = debounce((query) => fetchResults(query), 300);
// input.addEventListener("input", e => search(e.target.value))
// Only fires 300ms after user stops typing

// ── Throttle — limit to once per interval ────────────────────
function throttle(fn, limit) {
  let active = false;
  return (...args) => {
    if (active) return;
    active = true;
    fn(...args);
    setTimeout(() => active = false, limit);
  };
}
// window.addEventListener("scroll", throttle(onScroll, 100));
// Max once per 100ms regardless of scroll speed
// 02
Callbacks & Callback Hell
Callbacks were the original async pattern — a function passed as an argument, called when async work completes. They work but nest unreadably at scale.
callbacks.js 
// ── Basic callback pattern ────────────────────────────────────
function fetchUser(id, callback) {
  setTimeout(() => {
    const user = { id, name: "Alice" };
    callback(null, user);     // Node.js convention: (error, result)
  }, 500);
}

fetchUser(1, (err, user) => {
  if (err) { console.error(err); return; }
  console.log(user.name);  // "Alice"
});

// ── Callback Hell (Pyramid of Doom) ──────────────────────────
// Each step depends on the previous — nesting grows forever
fetchUser(1, (err, user) => {
  if (err) return handleErr(err);
  fetchPosts(user.id, (err, posts) => {       // ← indent level 2
    if (err) return handleErr(err);
    fetchComments(posts[0].id, (err, comments) => {  // ← level 3
      if (err) return handleErr(err);
      fetchLikes(comments[0].id, (err, likes) => {   // ← level 4
        if (err) return handleErr(err);
        console.log(likes);  // ← we finally got here
        // Each level: repeated error handling, growing indentation
        // Extremely hard to read, maintain, and reason about
      });
    });
  });
});

// ── Problems with callbacks ───────────────────────────────────
// 1. No standard error handling (each library does it differently)
// 2. Inversion of control — you give your callback to a 3rd party
//    they control when/if it's called, how many times, etc.
// 3. Nesting makes control flow hard to follow
// 4. Can't use try/catch for async errors
// 5. Hard to run things in parallel

// ── Flatter callbacks (partial improvement) ───────────────────
function handleErr(err) { console.error(err); }

function onUser(err, user) {
  if (err) return handleErr(err);
  fetchPosts(user.id, onPosts);
}
function onPosts(err, posts) {
  if (err) return handleErr(err);
  fetchComments(posts[0].id, onComments);
}
// Flatter but still awkward — Promises solve this properly
fetchUser(1, onUser);
// 03
JavaScript Promises
A Promise represents an eventual value — it wraps async operations and provides a cleaner, chainable API with proper error handling.

Promise states

A Promise is always in one of three states:
Pending — initial state, neither fulfilled nor rejected.
Fulfilled — the operation completed successfully, value available.
Rejected — the operation failed, reason (error) available.
Once settled (fulfilled or rejected), a Promise's state never changes.

promises.js 
// ── Creating a Promise ────────────────────────────────────────
const myPromise = new Promise((resolve, reject) => {
  // resolve(value) — fulfil the promise with a value
  // reject(reason) — reject the promise with an error

  const success = Math.random() > 0.5;
  setTimeout(() => {
    if (success) resolve("Data loaded!");
    else         reject(new Error("Network failed"));
  }, 1000);
});

// ── Consuming a Promise ───────────────────────────────────────
myPromise
  .then(value => {
    console.log("Success:", value);  // runs on resolve
  })
  .catch(err => {
    console.error("Error:", err.message);  // runs on reject
  })
  .finally(() => {
    console.log("Always runs");  // cleanup — runs either way
  });

// ── Wrapping callbacks in Promises ───────────────────────────
function fetchUser(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id <= 0) reject(new Error("Invalid ID"));
      else          resolve({ id, name: "Alice" });
    }, 500);
  });
}

// ── Static Promise methods ────────────────────────────────────

// Promise.resolve / Promise.reject — instant settled promises
const instant = Promise.resolve("immediate value");

// Promise.all — run in PARALLEL, wait for ALL to resolve
// Fails fast: if ANY rejects, the whole thing rejects
Promise.all([
  fetchUser(1),
  fetchUser(2),
  fetchUser(3)
]).then(([u1, u2, u3]) => {
  console.log(u1, u2, u3);  // all three users
}).catch(err => console.error(err));

// Promise.allSettled — wait for ALL, collect all results
// Does NOT fail fast — every result is captured
Promise.allSettled([
  fetchUser(1),
  fetchUser(-1),  // will reject
  fetchUser(3)
]).then(results => {
  results.forEach(r => {
    if (r.status === "fulfilled") console.log("✅", r.value);
    else                             console.log("❌", r.reason.message);
  });
});

// Promise.race — first to settle wins
Promise.race([
  fetchUser(1),
  new Promise((_, reject) => setTimeout(() => reject(new Error("Timeout")), 300))
]).then(console.log).catch(console.error);

// Promise.any — first to FULFILL wins (ignores rejections)
Promise.any([
  Promise.reject(new Error("fail 1")),
  Promise.resolve("success"),
  Promise.resolve("also success")
]).then(console.log);  // "success"
// 04
Promise Chaining
Each .then() returns a new Promise — enabling flat, readable chains instead of nested callbacks.
promise_chaining.js 
// ── The same nested callbacks from before — now flat ──────────
fetchUser(1)
  .then(user => fetchPosts(user.id))    // return a new Promise
  .then(posts => fetchComments(posts[0].id))
  .then(comments => fetchLikes(comments[0].id))
  .then(likes => console.log(likes))
  .catch(err => console.error(err));    // ONE catch handles ALL errors

// ── Key chaining rules ────────────────────────────────────────
// 1. return a value from .then() → next .then() gets it
// 2. return a Promise from .then() → chain waits for it to resolve
// 3. throw inside .then() → jumps to .catch()
// 4. return inside .catch() → recovers, resumes the chain
// 5. .finally() doesn't receive the value — just cleans up

Promise.resolve(1)
  .then(n => n + 1)           // 2 — pass value forward
  .then(n => n * 10)          // 20
  .then(n => { throw new Error("oops"); })  // jumps to catch
  .catch(err => {
    console.log("Caught:", err.message);  // "oops"
    return 99;                            // recovered — chain continues
  })
  .then(n => console.log("Final:", n))  // "Final: 99"
  .finally(() => console.log("Done"));

// ── Real-world chain: fetch user → posts → render ─────────────
function loadUserDashboard(userId) {
  return fetchUser(userId)
    .then(user => {
      renderHeader(user);
      return fetchPosts(user.id);       // returns Promise — chain waits
    })
    .then(posts => {
      renderPosts(posts);
      return Promise.all(               // parallel fetches
        posts.map(p => fetchComments(p.id))
      );
    })
    .then(allComments => {
      allComments.forEach(renderComments);
    })
    .catch(err => {
      renderError(err.message);
    })
    .finally(() => {
      hideLoadingSpinner();
    });
}
// 05
async & await
async/await is syntactic sugar over Promises — it lets you write asynchronous code that reads like synchronous code.

async/await is Promises in disguise

async before a function makes it always return a Promise. await inside an async function pauses execution until a Promise settles, then returns the resolved value. The function resumes from that line. Under the hood it's all still the event loop and Promises — await is just readable syntax for .then().

async_await.js 
// ── Basic async/await ─────────────────────────────────────────
async function loadUser(id) {
  const user = await fetchUser(id);     // pauses here, waits for Promise
  const posts = await fetchPosts(user.id);  // pauses again
  return { user, posts };
}

// async function always returns a Promise
loadUser(1).then(console.log);

// ── Error handling with try/catch ─────────────────────────────
async function safeFetch(url) {
  try {
    const response = await fetch(url);

    if (!response.ok) {
      throw new Error(`HTTP error: ${response.status}`);
    }

    const data = await response.json();
    return data;

  } catch (err) {
    console.error("Fetch failed:", err.message);
    throw err;  // re-throw so caller can handle too
  } finally {
    hideSpinner();  // always runs
  }
}

// ── Sequential vs Parallel ────────────────────────────────────

// ❌ SEQUENTIAL — runs one after the other (slow: 500ms + 500ms = 1000ms)
async function sequential() {
  const user1 = await fetchUser(1);  // waits 500ms
  const user2 = await fetchUser(2);  // waits another 500ms
  return [user1, user2];
}

// ✅ PARALLEL — both start immediately (fast: ~500ms total)
async function parallel() {
  const [user1, user2] = await Promise.all([
    fetchUser(1),
    fetchUser(2)
  ]);
  return [user1, user2];
}

// ── await in loops ────────────────────────────────────────────

// ❌ forEach doesn't await — all run simultaneously
[1,2,3].forEach(async id => {
  const user = await fetchUser(id);
  // ⚠️ forEach doesn't await these — fires all at once
});

// ✅ for...of — properly awaits each iteration
async function processInOrder(ids) {
  for (const id of ids) {
    const user = await fetchUser(id);  // awaits each in sequence
    process(user);
  }
}

// ✅ Promise.all in loop — parallel but collected
async function processAll(ids) {
  const users = await Promise.all(ids.map(fetchUser));
  users.forEach(process);
}
