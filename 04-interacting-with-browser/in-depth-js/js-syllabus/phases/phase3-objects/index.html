Objects in JavaScript
Objects are collections of key-value pairs. They are the fundamental data structure for modelling real-world entities.

What is an object?

An object is an unordered collection of properties. Each property has a key (string or Symbol) and a value (any type). Objects are mutable and stored by reference. They are the building blocks of OOP and the foundation of JSON, DOM nodes, arrays, and virtually everything in JS.

objects.js 
// ── Creating objects ──────────────────────────────────────────
// 1. Object literal (most common)
const user = {
  name: "Alice",          // string key (quotes optional if valid identifier)
  age: 25,
  "is-active": true,     // keys with hyphens NEED quotes
  address: {              // nested object
    city: "New York",
    zip: "10001"
  }
};

// 2. Using new Object()
const obj = new Object();
obj.name = "Bob";

// ── Accessing properties ──────────────────────────────────────
console.log(user.name);           // dot notation — use when key is known
console.log(user["name"]);        // bracket notation — for dynamic keys
console.log(user["is-active"]);   // bracket required for hyphenated keys
console.log(user.address.city);   // nested access

const key = "age";
console.log(user[key]);  // 25 — dynamic key access

// ── Modifying properties ──────────────────────────────────────
user.age = 26;              // update
user.email = "a@b.com";   // add new property
delete user["is-active"]; // remove property

// ── Computed property names (ES6) ─────────────────────────────
const prefix = "user";
const dynamic = {
  [prefix + "Name"]: "Alice",  // computed key: "userName"
  [Symbol("id")]: 42           // Symbol key
};

// ── Property shorthand (ES6) ──────────────────────────────────
const name = "Alice";
const age = 25;
const person = { name, age };  // same as { name: name, age: age }

// ── Checking properties ───────────────────────────────────────
console.log("name" in user);          // true — checks own + inherited
console.log(user.hasOwnProperty("name")); // true — checks own only
console.log(user.missing);            // undefined — no error for missing keys
// 01
Object References & Copying
Objects are stored by reference — understanding this prevents subtle bugs when copying or comparing them.

Primitives vs Reference types

Primitives (number, string, etc.) are copied by value — each variable holds its own independent copy. Objects are copied by reference — multiple variables can point to the same object in memory. Modifying through one reference modifies the shared object.

references.js 
// ── Primitive copy (by value) ─────────────────────────────────
let a = 10;
let b = a;   // b gets its own copy of 10
b = 20;
console.log(a); // 10 — unchanged

// ── Object copy (by reference) ────────────────────────────────
const user1 = { name: "Alice" };
const user2 = user1;  // ⚠️ both point to the SAME object
user2.name = "Bob";
console.log(user1.name); // "Bob" — user1 was mutated!
console.log(user1 === user2); // true — same reference

// ── Shallow copy (one level deep) ────────────────────────────
// Method 1: Object.assign
const user3 = Object.assign({}, user1);
user3.name = "Charlie";
console.log(user1.name); // "Bob" — user1 unchanged now ✅

// Method 2: spread operator (preferred)
const user4 = { ...user1 };
user4.name = "Dave";
console.log(user1.name); // "Bob" — unaffected ✅

// ── Shallow copy problem — nested objects still shared ────────
const original = { name: "Alice", address: { city: "NY" } };
const shallow = { ...original };
shallow.address.city = "LA";  // ⚠️ modifies original.address too!
console.log(original.address.city); // "LA" — nested reference still shared

// ── Deep copy (completely independent) ───────────────────────
// Method 1: JSON (simple but loses functions, dates, undefined)
const deep1 = JSON.parse(JSON.stringify(original));
deep1.address.city = "Chicago";
console.log(original.address.city); // "LA" — truly independent ✅

// Method 2: structuredClone (modern, preserves more types)
const deep2 = structuredClone(original);  // ES2022+

// ── Object equality ───────────────────────────────────────────
const x = { a: 1 };
const y = { a: 1 };
console.log(x === y);  // false — different references
console.log(x === x);  // true  — same reference
// To compare by content: JSON.stringify(x) === JSON.stringify(y)
// 02
Object Methods
Built-in static methods on the Object class for working with keys, values, entries, and property descriptors.
object_methods.js 
const person = { name: "Alice", age: 25, city: "NY" };

// ── Object.keys / values / entries ───────────────────────────
console.log(Object.keys(person));
// ["name", "age", "city"] — array of keys

console.log(Object.values(person));
// ["Alice", 25, "NY"] — array of values

console.log(Object.entries(person));
// [["name","Alice"], ["age",25], ["city","NY"]] — array of [key,val] pairs

// Iterating with entries — most useful pattern
for (const [key, value] of Object.entries(person)) {
  console.log(`${key}: ${value}`);
}

// ── Object.assign ─────────────────────────────────────────────
const defaults = { theme: "dark", lang: "en" };
const userPrefs = { lang: "fr" };
const config = Object.assign({}, defaults, userPrefs);
console.log(config);  // { theme: "dark", lang: "fr" } — userPrefs overrides

// ── Object.freeze / Object.seal ──────────────────────────────
const frozen = Object.freeze({ x: 1, y: 2 });
frozen.x = 99;   // ⚠️ silently fails (or TypeError in strict mode)
console.log(frozen.x); // 1 — unchanged
// freeze: no add, no delete, no modify
// seal:   no add, no delete, CAN modify existing

// ── Object.fromEntries ────────────────────────────────────────
const entries = [["a", 1], ["b", 2]];
const fromE = Object.fromEntries(entries);
console.log(fromE);  // { a: 1, b: 2 }

// Transform object via entries → fromEntries pipeline
const prices = { apple: 1.5, banana: 0.5, cherry: 3 };
const doubled = Object.fromEntries(
  Object.entries(prices).map(([k, v]) => [k, v * 2])
);
console.log(doubled); // { apple: 3, banana: 1, cherry: 6 }

// ── Destructuring objects ─────────────────────────────────────
const { name, age, city = "Unknown" } = person;
console.log(name, age, city);  // "Alice"  25  "NY"

// Rename while destructuring
const { name: fullName, age: years } = person;
console.log(fullName, years);  // "Alice"  25

Optional Chaining (?.) and Nullish Coalescing (??)
Safe navigation through potentially null/undefined values without defensive if-checks everywhere.
optional_chaining.js 
const user = {
  name: "Alice",
  address: { city: "New York", zip: "10001" },
  getAge: () => 25
};
const guest = null;

// ── Without optional chaining — verbose ───────────────────────
const city1 = user && user.address && user.address.city;  // old way

// ── With optional chaining ?. ─────────────────────────────────
const city2 = user?.address?.city;
console.log(city2);                  // "New York" ✅

const guestCity = guest?.address?.city;
console.log(guestCity);              // undefined (no error!) ✅

// ?. on methods
console.log(user?.getAge?.());      // 25
console.log(guest?.getAge?.());     // undefined (no TypeError)

// ?. with bracket notation
const prop = "city";
console.log(user?.address?.[prop]);  // "New York"

// ── Nullish coalescing ?? — provide fallback ──────────────────
const port = null;
console.log(port ?? 3000);     // 3000 — null triggers ??
console.log(0 ?? 3000);        // 0 — 0 does NOT trigger ?? (unlike ||)
console.log("" ?? "default");  // "" — empty string does NOT trigger ??

// ── Combining ?. and ?? ───────────────────────────────────────
const username = guest?.profile?.name ?? "Anonymous";
console.log(username);  // "Anonymous" — safe navigation + fallback

// ── Nullish assignment ??= ────────────────────────────────────
let config = null;
config ??= { theme: "dark" };  // only assigns if null/undefined
console.log(config);            // { theme: "dark" }
// 04
JSON — JavaScript Object Notation
JSON is a text format for data exchange. It looks like JS objects but is a string — stricter syntax rules apply.
json.js 
// ── JSON rules ────────────────────────────────────────────────
// Keys MUST be double-quoted strings
// Values: string, number, boolean, null, array, object
// No: undefined, functions, Symbol, Infinity, NaN, comments
// No trailing commas

const jsonString = '{"name":"Alice","age":25,"active":true}';

// ── JSON.parse — string → JS object ──────────────────────────
const obj = JSON.parse(jsonString);
console.log(obj.name);   // "Alice"
console.log(obj.age);    // 25 (number, not string)

// JSON.parse with reviver (transform values during parsing)
const dateJson = '{"date":"2026-01-15","value":42}';
const parsed = JSON.parse(dateJson, (key, val) => {
  if (key === "date") return new Date(val);  // convert string to Date
  return val;
});
console.log(parsed.date instanceof Date);  // true

// ── JSON.stringify — JS object → string ──────────────────────
const user = { name: "Alice", age: 25, fn: () => {} };
console.log(JSON.stringify(user));
// '{"name":"Alice","age":25}'  — function is dropped!

// With formatting (indent)
console.log(JSON.stringify(user, null, 2));
// Formatted with 2-space indentation — great for debugging

// JSON.stringify with replacer (select which keys to include)
console.log(JSON.stringify(user, ["name"]));
// '{"name":"Alice"}' — only name key included

// ── What JSON loses ───────────────────────────────────────────
const complex = {
  fn: () => {},             // dropped
  undef: undefined,          // dropped
  sym: Symbol("x"),          // dropped
  date: new Date(),           // becomes string
  map: new Map([["a",1]]),    // becomes {}
  nan: NaN,                   // becomes null
  inf: Infinity               // becomes null
};

Symbol Type
Symbols are unique, immutable primitive values. They are used as guaranteed-unique property keys.
symbols.js 
// ── Creating Symbols ──────────────────────────────────────────
const id1 = Symbol("id");  // "id" is a description (for debugging only)
const id2 = Symbol("id");
console.log(id1 === id2);  // false — EVERY Symbol is unique
console.log(typeof id1);   // "symbol"

// ── Symbol as object key ──────────────────────────────────────
const ID = Symbol("id");
const user = {
  name: "Alice",
  [ID]: 42    // Symbol key — hidden from for..in, Object.keys, JSON
};
console.log(user[ID]);       // 42 — access with the symbol variable
console.log(Object.keys(user)); // ["name"] — Symbol not included
// JSON.stringify(user) also ignores Symbol keys

// ── Symbol.for — global registry ─────────────────────────────
const s1 = Symbol.for("shared");
const s2 = Symbol.for("shared");
console.log(s1 === s2);  // true — reuses from global registry

// ── Well-known Symbols (built-in hooks) ───────────────────────
// Symbol.iterator — makes an object iterable
const range = {
  from: 1, to: 5,
  [Symbol.iterator]() {
    let current = this.from;
    const last = this.to;
    return {
      next() {
        return current <= last
          ? { value: current++, done: false }
          : { done: true };
      }
    };
  }
};
for (const n of range) console.log(n);  // 1 2 3 4 5

Strings
Strings are immutable sequences of characters. They have a rich built-in API for searching, transforming, and formatting text.
strings.js 
const str = "Hello, World!";

// ── Properties ────────────────────────────────────────────────
console.log(str.length);       // 13
console.log(str[0]);          // "H"
console.log(str.at(-1));      // "!" — negative index from end

// ── Searching ─────────────────────────────────────────────────
console.log(str.includes("World"));    // true
console.log(str.startsWith("Hello"));  // true
console.log(str.endsWith("!"));        // true
console.log(str.indexOf("o"));         // 4 — first occurrence
console.log(str.lastIndexOf("o"));     // 8 — last occurrence

// ── Extracting ────────────────────────────────────────────────
console.log(str.slice(7, 12));    // "World"  (start, end)
console.log(str.slice(-6));       // "orld!"  negative = from end
console.log(str.substring(7, 12)); // "World"  (can't use negative)

// ── Transforming ──────────────────────────────────────────────
console.log(str.toUpperCase());        // "HELLO, WORLD!"
console.log(str.toLowerCase());        // "hello, world!"
console.log("  hi  ".trim());          // "hi" — removes whitespace both ends
console.log("  hi  ".trimStart());     // "hi  " — leading only
console.log(str.replace("World", "JS")); // "Hello, JS!" — first match
console.log(str.replaceAll("l", "L"));  // "HeLLo, WorLd!" — all matches

// ── Splitting & Joining ───────────────────────────────────────
const parts = "a,b,c,d".split(",");
console.log(parts);                   // ["a","b","c","d"]
console.log(parts.join(" - "));       // "a - b - c - d"
console.log([..."hello"]);           // ["h","e","l","l","o"] — spread string

// ── Padding & Repeating ───────────────────────────────────────
console.log("5".padStart(3, "0"));  // "005" — pad to length 3
console.log("hi".repeat(3));        // "hihihi"

// ── Template literals ─────────────────────────────────────────
const name = "Alice", age = 25;
console.log(`My name is ${name} and I am ${age} years old`);
// Expressions inside ${}: ${2 + 2}, ${name.toUpperCase()}

// Multi-line strings
const multiline = `Line 1
Line 2
Line 3`;

// Tagged template literals
function highlight(strings, ...values) {
  return strings.reduce((acc, str, i) =>
    acc + str + (values[i] ? `[${values[i]}]` : ""), "");
}
console.log(highlight`Hello ${name}, you are ${age}`);
// "Hello [Alice], you are [25]"

Arrays & Array Methods
Arrays are ordered, zero-indexed lists. JavaScript's array API is one of the richest in any language.
arrays.js 
const nums = [3, 1, 4, 1, 5, 9, 2, 6];

// ── MUTATING METHODS (change original array) ──────────────────
const arr = [1, 2, 3];
arr.push(4);         // [1,2,3,4]     — add to end
arr.pop();           // [1,2,3]       — remove from end
arr.unshift(0);      // [0,1,2,3]     — add to start
arr.shift();         // [1,2,3]       — remove from start
arr.splice(1, 1);    // [1,3]         — remove 1 item at index 1
arr.splice(1, 0, 9); // [1,9,3]       — insert 9 at index 1
arr.sort((a,b) => a-b); // [1,3,9]    — sort numerically ascending
arr.reverse();       // [9,3,1]       — reverse in-place

// ── NON-MUTATING (return new array/value) ─────────────────────

// map — transform each element
const doubled = nums.map(n => n * 2);
console.log(doubled); // [6,2,8,2,10,18,4,12]

// filter — keep elements matching condition
const evens = nums.filter(n => n % 2 === 0);
console.log(evens);   // [4,2,6]

// reduce — accumulate into single value
const total = nums.reduce((acc, n) => acc + n, 0);
console.log(total);   // 31

// find / findIndex — first match
console.log(nums.find(n => n > 4));      // 5 — first element > 4
console.log(nums.findIndex(n => n > 4)); // 4 — index of first element > 4

// some / every — boolean checks
console.log(nums.some(n => n > 8));    // true  — at least one > 8
console.log(nums.every(n => n > 0));   // true  — all > 0

// includes — check for value
console.log(nums.includes(9));          // true

// flat / flatMap
const nested = [[1,2],[3,[4,5]]];
console.log(nested.flat());    // [1,2,3,[4,5]] — one level
console.log(nested.flat(2));   // [1,2,3,4,5]   — two levels
console.log(nested.flat(Infinity)); // fully flatten

// slice — extract without mutating
console.log(nums.slice(1, 4));  // [1,4,1]

// Array.from — create array from iterable or array-like
console.log(Array.from("hello"));         // ["h","e","l","l","o"]
console.log(Array.from({length:3}, (_,i) => i*2)); // [0,2,4]

// Spread to combine arrays
const combined = [...[1,2], ...[3,4]];   // [1,2,3,4]
map → filter → reduce pipeline (functional style)
▶nums.filter(n => n > 3).map(n => n * 2).reduce((a,b) => a+b, 0)
▶→ 50 (chain multiple operations without mutating)
