Closures & Lexical Scoping — Deep Dive
Closures aren't just a concept — they're the mechanism behind modules, memoization, currying, and private state.

The formal definition, with depth

A closure is a function bundled together with references to its surrounding lexical environment. "Lexical" means where the function was written, not where it's called. The environment is captured at the time the inner function is defined — it's a snapshot of variable bindings that the inner function carries with it forever, even after the outer function has returned and its call frame is gone from the stack.

closures_deep.js 
// ── Memoization — cache results via closure ───────────────────
function memoize(fn) {
  const cache = new Map();  // closed over — persists between calls
  return (...args) => {
    const key = JSON.stringify(args);
    if (cache.has(key)) { console.log("hit"); return cache.get(key); }
    const result = fn(...args);
    cache.set(key, result);
    return result;
  };
}
const fib = memoize(n => n <= 1 ? n : fib(n-1) + fib(n-2));
fib(40); // computed
fib(40); // "hit" — instant

// ── Currying — one argument at a time ─────────────────────────
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn(...args);
    return (...more) => curried(...args, ...more);
    // ↑ closes over accumulated args — returns new function
  };
}
const add = curry((a, b, c) => a + b + c);
console.log(add(1)(2)(3));  // 6
const add5 = add(5);         // partial application
console.log(add5(10)(2));    // 17

// ── Module pattern via IIFE + closure ─────────────────────────
const bankAccount = (function() {
  let balance = 0;  // private — not accessible outside
  return {
    deposit(n)    { balance += n; },
    withdraw(n)   { balance -= n; },
    getBalance()  { return balance; }
  };
})();
bankAccount.deposit(100);
bankAccount.withdraw(30);
console.log(bankAccount.getBalance());  // 70
// console.log(balance);                // ❌ ReferenceError — truly private

// ── Loop closure — the classic gotcha ─────────────────────────
// var: ONE binding for the whole loop
const funcsVar = [];
for (var i = 0; i < 3; i++) funcsVar.push(() => i);
console.log(funcsVar.map(f => f()));  // [3, 3, 3] — all see same i

// let: new binding PER iteration
const funcsLet = [];
for (let i = 0; i < 3; i++) funcsLet.push(() => i);
console.log(funcsLet.map(f => f()));  // [0, 1, 2] ✅ each has own i

Understanding "this" in Every Context
this is determined by HOW a function is called, not WHERE it is written. Six distinct contexts.

The four binding rules — in priority order

1. new — new Fn() → this = new empty object.
2. Explicit — .call() / .apply() / .bind() → this = first arg.
3. Implicit — obj.method() → this = obj.
4. Default — plain fn() → window (sloppy) or undefined (strict).
Arrow exception — arrow functions ignore ALL rules; they inherit this lexically.

this_contexts.js 
"use strict";

// 1. Global context
console.log(this);  // window (browser) / {} (Node module)

// 2. Regular function call — default binding
function show() { console.log(this); }
show();  // undefined (strict) / window (sloppy)

// 3. Method call — implicit binding (this = obj before the dot)
const user = { name: "Alice", greet() { return this.name; } };
user.greet();  // "Alice"

// Implicit binding LOST when method is extracted
const fn = user.greet;
fn();  // undefined — this is no longer user (default binding)

// 4. Arrow function — lexical this (from enclosing scope)
const obj = {
  name: "obj",
  regular() {
    const inner = () => this.name;  // captures this from regular()
    return inner();
  },
  arrow: () => this.name  // ⚠️ captures this from module scope — undefined
};
obj.regular(); // "obj" ✅
obj.arrow();   // undefined ⚠️ — arrow as method is wrong

// 5. Class — this = instance
class Counter {
  count = 0;
  increment()  { this.count++; }           // this lost as callback
  safeInc = () => { this.count++; }        // arrow field — always bound
}
const c = new Counter();
const { increment, safeInc } = c;
// increment();  ❌ this = undefined
// safeInc();    ✅ this = c instance

// 6. Event handler
// function handler: this = the element
// arrow handler:    this = outer lexical scope (not the element)
this rule priority (highest → lowest)
▶new keyword → new object
▶call / apply / bind → first argument
▶obj.method() → obj
▶fn() → window / undefined
▶arrow → ignores all rules, lexical this always

call, apply, and bind
Three methods to explicitly set this. Essential for method borrowing, fixing lost context, and partial application.
call_apply_bind.js 
function introduce(greeting, punct) {
  return `${greeting}, I'm ${this.name}${punct}`;
}
const alice = { name: "Alice" };
const bob   = { name: "Bob"   };

// ── .call(thisArg, arg1, arg2) — call NOW, args as list ───────
console.log(introduce.call(alice, "Hello", "!"));  // "Hello, I'm Alice!"
console.log(introduce.call(bob,   "Hi",    "."));  // "Hi, I'm Bob."

// ── .apply(thisArg, [argsArray]) — call NOW, args as array ────
console.log(introduce.apply(alice, ["Hey", "?"]));  // "Hey, I'm Alice?"

// apply use-case: spread array as args (pre-spread era)
const nums = [3, 1, 4, 1, 5];
Math.max.apply(null, nums);  // same as Math.max(...nums) today

// ── .bind(thisArg, arg1) — returns NEW function, doesn't call ─
const aliceIntro = introduce.bind(alice);
console.log(aliceIntro("Yo", "!"));   // "Yo, I'm Alice!" — reusable

// Partial application with bind — pre-fill arguments
const aliceHello = introduce.bind(alice, "Hello");
console.log(aliceHello("!"));  // "Hello, I'm Alice!"
console.log(aliceHello("."));  // "Hello, I'm Alice."

// ── Fixing lost this with bind ────────────────────────────────
class Timer {
  constructor() {
    this.seconds = 0;
    this.tick = this.tick.bind(this);  // bind in constructor
  }
  tick() { this.seconds++; console.log(this.seconds); }
}
const t = new Timer();
setInterval(t.tick, 1000);  // ✅ this is bound — works as callback

// ── Method borrowing ──────────────────────────────────────────
const arrayLike = { 0: "a", 1: "b", length: 2 };
const arr = Array.prototype.slice.call(arrayLike);
console.log(arr);  // ["a", "b"] — borrowed Array method

//  call   → call NOW, args comma-separated
//  apply  → call NOW, args as array
//  bind   → return NEW bound function, call later

Error Handling with try-catch
Robust programs anticipate failures. JS error handling for both sync and async code.
error_handling.js 
// ── try / catch / finally ─────────────────────────────────────
try {
  const data = JSON.parse("{ bad json");
} catch (err) {
  console.log(err.name);     // "SyntaxError"
  console.log(err.message);  // descriptive message
  console.log(err.stack);    // full stack trace string
} finally {
  console.log("always runs — use for cleanup");
}

// ── Built-in error types ──────────────────────────────────────
// Error          → generic base
// SyntaxError    → JSON.parse("bad"), eval()
// TypeError      → null.property, undefined()
// RangeError     → new Array(-1), number out of range
// ReferenceError → accessing undeclared variable
// URIError       → malformed URI component

// ── Catch specific types ───────────────────────────────────────
try {
  riskyOperation();
} catch (err) {
  if      (err instanceof TypeError)  handleTypeError(err);
  else if (err instanceof RangeError) handleRangeError(err);
  else    throw err;  // ← re-throw unexpected — don't swallow
}

// ── Async error handling — try/catch with await ───────────────
async function loadData() {
  try {
    const res  = await fetch("/api/data");
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } catch (err) {
    console.error("Load failed:", err.message);
    return null;  // graceful fallback
  } finally {
    hideSpinner();
  }
}

// ── Global safety nets ────────────────────────────────────────
window.addEventListener("error", e => {
  console.error("Uncaught:", e.error);
  sendToMonitoring(e.error);
});
window.addEventListener("unhandledrejection", e => {
  console.error("Unhandled Promise rejection:", e.reason);
});
// 04
Custom Errors
Extend Error to create domain-specific, instanceof-checkable errors with structured data.

Why custom errors matter

Generic errors carry only a message string. Custom errors attach structured data (status codes, field names, context), are identifiable by type via instanceof, and let you build error hierarchies that mirror your domain. Callers catch exactly what they can handle and re-throw the rest.

custom_errors.js 
// ── Base custom error ─────────────────────────────────────────
class AppError extends Error {
  constructor(message, code, context = {}) {
    super(message);
    this.name    = this.constructor.name;
    this.code    = code;
    this.context = context;
    Object.setPrototypeOf(this, new.target.prototype);
  }
}

// ── Domain-specific subclasses ────────────────────────────────
class ValidationError extends AppError {
  constructor(field, message, value) {
    super(message, "VALIDATION_ERROR", { field, value });
    this.field = field;
  }
}
class NetworkError extends AppError {
  constructor(message, status, url) {
    super(message, "NETWORK_ERROR", { status, url });
    this.status = status;
  }
}
class NotFoundError extends NetworkError {
  constructor(url) { super(`Not found: ${url}`, 404, url); }
}
class AuthError extends AppError {
  constructor(msg = "Unauthorized") { super(msg, "AUTH_ERROR"); }
}

// ── Throwing ──────────────────────────────────────────────────
function validateAge(age) {
  if (typeof age !== "number")
    throw new ValidationError("age", "Must be a number", age);
  if (age < 0 || age > 150)
    throw new ValidationError("age", "Out of valid range", age);
}

// ── Catching by type ──────────────────────────────────────────
async function loadUser(id) {
  try {
    return await fetchUser(id);
  } catch (err) {
    if (err instanceof NotFoundError)  return null;
    if (err instanceof AuthError)       return redirectToLogin();
    if (err instanceof ValidationError) return showFieldError(err.field, err.message);
    throw err;  // unknown — re-throw
  }
}

// ── instanceof walks the full hierarchy ───────────────────────
const e = new NotFoundError("/api/user/1");
console.log(e instanceof NotFoundError);  // true
console.log(e instanceof NetworkError);   // true
console.log(e instanceof AppError);       // true
console.log(e instanceof Error);           // true
console.log(e.name);    // "NotFoundError"
console.log(e.status);  // 404
console.log(e.code);    // "NETWORK_ERROR"
error hierarchy
▶Error (built-in)
▶ └── AppError
▶ ├── ValidationError (field, value)
▶ ├── NetworkError (status, url)
▶ │ └── NotFoundError
▶ └── AuthError
