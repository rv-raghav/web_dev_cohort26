What is a Function?
A function is a reusable block of code designed to perform a specific task. Functions are first-class citizens in JavaScript.

First-class functions

In JavaScript, functions are first-class values — they can be:
• Assigned to variables
• Passed as arguments to other functions
• Returned from other functions
• Stored in arrays and objects
This enables functional programming patterns like higher-order functions, callbacks, and closures.

Functional Programming Paradigm

FP treats computation as the evaluation of mathematical functions. Key ideas: pure functions (same input always produces same output, no side effects), immutability (don't mutate data, return new data), higher-order functions (functions that take or return functions). JS supports both OOP and FP styles — use the right tool for the context.

functions_intro.js 
// ── Functions are values ──────────────────────────────────────

// 1. Assign to variable
const greet = function() { return "hello"; };

// 2. Pass as argument (callback)
[1,2,3].forEach(function(n) { console.log(n); });

// 3. Return from function (higher-order function)
function makeMultiplier(factor) {
  return function(n) {     // ← returns a function!
    return n * factor;
  };
}
const double = makeMultiplier(2);
console.log(double(5));  // 10
console.log(double(8));  // 16

// 4. Store in object
const obj = {
  name: "Alice",
  sayHi: function() { return `Hi, I'm ${this.name}`; }
};
console.log(obj.sayHi()); // "Hi, I'm Alice"

// Pure function example — no side effects
function add(a, b) { return a + b; }  // same input → same output always

// Impure function — has side effects
let count = 0;
function increment() { count++; }  // modifies external state — impure

Three Types of Functions
Declaration, Expression, and Arrow Function — they look different and behave differently in key ways.
three_types.js 
// ── TYPE 1: Function Declaration ──────────────────────────────
// Starts with the  function  keyword at the statement level
// FULLY HOISTED — can be called before it appears in code

sayHello();  // ✅ Works — hoisted

function sayHello() {
  console.log("Hello from declaration");
}


// ── TYPE 2: Function Expression ───────────────────────────────
// A function assigned as a VALUE to a variable
// Hoisted as undefined — cannot call before assignment

// sayBye();  ❌ TypeError: sayBye is not a function

const sayBye = function() {
  console.log("Bye from expression");
};

sayBye();  // ✅ Works after assignment

// Named function expression — useful for recursion + stack traces
const factorial = function fact(n) {
  return n <= 1 ? 1 : n * fact(n - 1);  // can call itself by name
};
console.log(factorial(5)); // 120


// ── TYPE 3: Arrow Function (ES6) ──────────────────────────────
// Concise syntax. Key differences from regular functions:
// 1. No own  this  binding
// 2. Cannot be used as constructor (no new)
// 3. No  arguments  object
// 4. Hoisted as undefined (like expressions)

const add = (a, b) => a + b;  // implicit return — no {} needed
console.log(add(3, 4));  // 7

const square = n => n * n;  // single param — no parens needed
console.log(square(5));     // 25

const greet = () => "Hello!";  // no params — empty parens required

const buildUser = (name) => ({ name, active: true });
// ↑ returning an object literal: wrap in () to avoid {} being
//   treated as function body

// Full body arrow (when multiple statements needed)
const processScore = (score) => {
  const grade = score >= 70 ? "pass" : "fail";
  return { score, grade };  // explicit return needed with {}
};
Feature Declaration Expression Arrow
Hoisted Fully undefined undefined
Own this Yes Yes No — inherits
arguments object Yes Yes No
Use as constructor Yes Yes No
Implicit return No No Yes (single expr)
Best for Top-level fns Callbacks Callbacks, methods

Parameters & Arguments
Parameters are placeholders in the definition. Arguments are the actual values passed at call time.
params_args.js 
// ── Basic parameters ──────────────────────────────────────────
function greet(name, greeting) {  // name, greeting = PARAMETERS
  console.log(`${greeting}, ${name}!`);
}
greet("Alice", "Hello");  // "Alice", "Hello" = ARGUMENTS
greet("Bob");              // greeting is undefined if not passed

// ── Default parameters (ES6) ──────────────────────────────────
function greet2(name, greeting = "Hello") {
  console.log(`${greeting}, ${name}!`);
}
greet2("Alice");             // "Hello, Alice!" — uses default
greet2("Alice", "Hi");       // "Hi, Alice!" — overrides default
greet2("Alice", undefined); // "Hello, Alice!" — undefined triggers default
greet2("Alice", null);      // "null, Alice!" — null does NOT trigger default

// ── Rest parameters (...args) ─────────────────────────────────
// Collects remaining arguments into a real Array
function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}
console.log(sum(1, 2, 3, 4, 5));  // 15

function first(a, b, ...rest) {
  console.log(a, b, rest);  // 1, 2, [3, 4, 5]
}
first(1, 2, 3, 4, 5);
// Rest must be the LAST parameter

// ── arguments object (old way — not in arrow fns) ─────────────
function oldSum() {
  let total = 0;
  for (let i = 0; i < arguments.length; i++) {
    total += arguments[i];
  }
  return total;
}
// arguments is array-LIKE, not a real Array — use rest params instead

// ── Destructured parameters ───────────────────────────────────
function displayUser({ name, age, city = "Unknown" }) {
  console.log(`${name}, ${age}, ${city}`);
}
displayUser({ name: "Alice", age: 25 });  // "Alice, 25, Unknown"

Returning from Functions
The return statement exits a function and optionally sends a value back to the caller.
return_values.js 
// ── Basic return ──────────────────────────────────────────────
function add(a, b) {
  return a + b;     // exits + sends value
  console.log("never runs");  // unreachable code after return
}
const result = add(3, 4);
console.log(result);  // 7

// No return = returns undefined
function doNothing() {}
console.log(doNothing()); // undefined

// ── Early return pattern ──────────────────────────────────────
// Return early to avoid deep nesting — "guard clauses"

// ❌ Nested version (hard to read)
function processNested(user) {
  if (user) {
    if (user.name) {
      if (user.age > 18) {
        return `Welcome ${user.name}`;
      }
    }
  }
  return "Invalid";
}

// ✅ Guard clause version (clean)
function process(user) {
  if (!user) return "Invalid";
  if (!user.name) return "Invalid";
  if (user.age <= 18) return "Invalid";
  return `Welcome ${user.name}`;
}

// ── Returning multiple values ─────────────────────────────────
// JS can only return ONE value — use array or object

function getMinMax(arr) {
  return [Math.min(...arr), Math.max(...arr)];  // return array
}
const [min, max] = getMinMax([3, 1, 7, 2]);
console.log(min, max);  // 1  7

function getStats(arr) {
  return {          // return object — named properties
    min: Math.min(...arr),
    max: Math.max(...arr),
    avg: arr.reduce((a, b) => a + b) / arr.length
  };
}
const { min: lo, max: hi, avg } = getStats([3, 1, 7, 2]);
console.log(lo, hi, avg);  // 1  7  3.25

Hoisting in Functions
The memory creation phase makes declarations available before execution. Function declarations are fully hoisted; expressions are not.

Two-phase execution model

Phase 1 — Memory Creation: JS scans the file before running a single line. Function declarations → full body stored. var variables → set to undefined. let/const → placed in Temporal Dead Zone (uninitialized).

Phase 2 — Execution: Code runs line by line. Values are assigned. Function calls create new execution contexts on the call stack.

hoisting_functions.js 
// ── PHASE 1 memory snapshot ──────────────────────────────────
//   declaration  →  ƒ () { console.log("declaration") }  ✅ full body
//   expression   →  undefined
//   arrow        →  undefined

// ── Can call before definition (declaration only) ─────────────
declaration();   // ✅ "I am a declaration" — fully hoisted
// expression();  // ❌ TypeError: expression is not a function
// arrow();       // ❌ TypeError: arrow is not a function

function declaration() {
  console.log("I am a declaration");
}

var expression = function() {
  console.log("I am an expression");
};

var arrow = () => {
  console.log("I am an arrow");
};

// After declarations — all three work fine
declaration();  // ✅
expression();   // ✅
arrow();        // ✅

// ── Function hoisting inside blocks ──────────────────────────
// ⚠️ Function declarations inside blocks are NOT reliably hoisted
if (true) {
  function blockFn() { return "in block"; }
}
// Behaviour varies between strict mode and sloppy mode
// Use function expressions inside blocks to be safe

Variable Scope
Scope determines where a variable is accessible. JavaScript has global, function, and block scope.

Lexical scoping

JavaScript uses lexical (static) scoping — a variable's scope is determined by where it's written in the code, not where it's called. Inner scopes can access variables from outer scopes, but not the other way around. This is the foundation of closures.

scope.js 
// ── Global scope ──────────────────────────────────────────────
const globalVar = "I'm global";  // accessible everywhere

function outer() {

  // ── Function scope ─────────────────────────────────────────
  const outerVar = "I'm in outer";

  console.log(globalVar);  // ✅ can access global
  console.log(outerVar);   // ✅ own scope

  function inner() {

    // ── Inner function scope ───────────────────────────────────
    const innerVar = "I'm in inner";

    console.log(globalVar);  // ✅ can access — outer scope
    console.log(outerVar);   // ✅ can access — outer scope
    console.log(innerVar);   // ✅ own scope
  }

  inner();
  // console.log(innerVar);  ❌ ReferenceError — innerVar not accessible here
}
outer();
// console.log(outerVar);  ❌ ReferenceError — not accessible globally


// ── Block scope (let/const) ───────────────────────────────────
{
  let blockLet = "let in block";
  var blockVar = "var in block";
}
// console.log(blockLet);  ❌ ReferenceError — let is block scoped
console.log(blockVar);    // ✅ "var in block" — var leaks out of blocks!


// ── Scope chain (how JS resolves variable names) ──────────────
const x = "global x";

function findX() {
  // no local x — walks up the chain
  console.log(x);  // "global x" — found in global scope
}

function findX2() {
  const x = "local x";  // shadows the global x
  console.log(x);         // "local x" — finds own scope first
}

findX();   // "global x"
findX2();  // "local x"

Closures
A closure is a function that remembers its lexical environment — the variables in scope at the time it was created — even after that outer function has returned.

The formal definition

A closure is the combination of a function and the lexical environment within which that function was declared. Every function in JavaScript forms a closure — it "closes over" the variables in its scope. This is how functions can remember values from their enclosing scope even when called later from a completely different context.

closures.js 
// ── Basic closure ─────────────────────────────────────────────
function makeCounter() {
  let count = 0;            // this variable is "closed over"

  return function() {       // returned function has access to count
    count++;
    return count;
  };
}

const counter1 = makeCounter();
const counter2 = makeCounter();

console.log(counter1());  // 1 — own count
console.log(counter1());  // 2
console.log(counter1());  // 3
console.log(counter2());  // 1 — separate instance, own count
// count is not accessible from outside — private data!


// ── Real-world use: function factory ─────────────────────────
function multiplier(factor) {
  return (n) => n * factor;  // closes over factor
}
const double = multiplier(2);
const triple = multiplier(3);
console.log(double(5));  // 10
console.log(triple(5)); // 15


// ── Module pattern using closure ──────────────────────────────
const bankAccount = (function() {
  let balance = 0;  // private — not accessible outside

  return {
    deposit(amount)  { balance += amount; },
    withdraw(amount) { balance -= amount; },
    getBalance()     { return balance; }
  };
})();  // IIFE — immediately invoked

bankAccount.deposit(100);
bankAccount.withdraw(30);
console.log(bankAccount.getBalance());  // 70
// console.log(balance);                 ❌ ReferenceError — truly private


// ── Classic closure gotcha in loops ──────────────────────────

// ❌ var in loop — all share the SAME i
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log("var:", i), 0);
}
// prints: var: 3  var: 3  var: 3  ← all closures share same i

// ✅ let in loop — each iteration gets its OWN block-scoped i
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log("let:", i), 0);
}
// prints: let: 0  let: 1  let: 2  ← each has its own closure
closure memory model
▶counter1() → count lives in counter1's closure scope
▶counter2() → completely separate count in its own closure
▶var loop: 3 3 3← closures share same var
▶let loop: 0 1 2← each iteration has own let binding
