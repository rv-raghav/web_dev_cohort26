// 00
The DOM Tree
The Document Object Model is a tree-shaped representation of your HTML. Every tag becomes a node. JS can read and modify any of it.

What the DOM is

When the browser parses HTML it creates a tree of Node objects. The document object is the entry point — it represents the entire page. Every HTML element is an Element node. Text inside elements is a Text node. Comments are Comment nodes. JavaScript can traverse, modify, create, and delete any node in the tree — and changes are reflected live in the browser.

dom_tree.js 
// ── The tree structure ────────────────────────────────────────
//
//   document
//   └── <html>
//       ├── <head>
//       │   └── <title> "My Page" </title>
//       └── <body>
//           ├── <h1> "Hello" </h1>
//           └── <div class="container">
//               ├── <p> "Text" </p>
//               └── <button id="btn"> "Click" </button>
//

// ── document object ───────────────────────────────────────────
console.log(document.title);       // "My Page"
console.log(document.URL);         // current page URL
console.log(document.body);        // the <body> element
console.log(document.head);        // the <head> element
console.log(document.documentElement); // the <html> element

// ── Node relationships ────────────────────────────────────────
const body = document.body;
console.log(body.parentNode);      // <html>
console.log(body.children);        // HTMLCollection of child elements
console.log(body.firstElementChild); // first child element
console.log(body.lastElementChild);  // last child element
console.log(body.nextElementSibling);// next sibling element

// ── Node types ────────────────────────────────────────────────
// nodeType:  1 = Element, 3 = Text, 8 = Comment, 9 = Document
console.log(document.nodeType);   // 9
console.log(body.nodeType);       // 1
console.log(body.nodeName);       // "BODY"
// 01
Methods for Accessing DOM Elements
Multiple ways to find elements — by ID, class, tag, CSS selector. Know which to use and why.
dom_access.js 
// ── getElementById — fastest, returns single element ──────────
const btn = document.getElementById("btn");
// Returns Element or null. No # prefix.

// ── querySelector — CSS selector, first match ─────────────────
const h1       = document.querySelector("h1");
const byId     = document.querySelector("#btn");
const byClass  = document.querySelector(".container");
const nested   = document.querySelector(".container p");
const attr     = document.querySelector("input[type='email']");
// Accepts any valid CSS selector — very flexible

// ── querySelectorAll — all matches, returns NodeList ──────────
const allPs  = document.querySelectorAll("p");
const cards  = document.querySelectorAll(".card");

// NodeList is NOT a real Array — convert to iterate with all Array methods
const arr = [...allPs];  // spread into Array
arr.forEach(p => console.log(p.textContent));
// querySelectorAll NodeList supports forEach directly (but not map/filter)

// ── getElementsByClassName / getElementsByTagName ─────────────
const live1 = document.getElementsByClassName("card");
const live2 = document.getElementsByTagName("p");
// These return LIVE HTMLCollections — auto-update when DOM changes
// querySelector returns STATIC NodeList — does not auto-update

// ── Search within an element (not just document) ──────────────
const nav = document.querySelector("nav");
const navLinks = nav.querySelectorAll("a");  // only links inside nav

// ── closest() — walk UP the tree to find ancestor ────────────
const item = document.querySelector(".item");
const list = item.closest(".list");  // finds nearest .list ancestor

// ── matches() — test if element matches a selector ────────────
if (btn.matches(".primary")) { console.log("is primary button"); }
when to use what
▶getElementById← fastest when you have an ID
▶querySelector← most flexible, use this for complex selectors
▶querySelectorAll← when you need multiple elements
// 02
Updating DOM Content
Change text, HTML, attributes, styles, and CSS classes on elements.
dom_update.js 
const el = document.querySelector("#output");

// ── Text content ──────────────────────────────────────────────
el.textContent = "Hello World";
// Safest — escapes HTML, treats value as plain text
// Use this for user-provided content to prevent XSS

// ── innerHTML ─────────────────────────────────────────────────
el.innerHTML = "<strong>Bold text</strong>";
// Parses HTML — powerful but dangerous with user input (XSS risk)
// ❌ NEVER: el.innerHTML = userInput  — XSS vulnerability

// ── Attributes ────────────────────────────────────────────────
const link = document.querySelector("a");
link.setAttribute("href", "https://example.com");
console.log(link.getAttribute("href"));  // "https://example.com"
link.removeAttribute("target");

// Property shorthand for common attributes
link.href  = "https://example.com";
link.id    = "main-link";
link.title = "Click me";

// dataset — access data-* attributes
// <div data-user-id="42" data-role="admin">
const div = document.querySelector("[data-user-id]");
console.log(div.dataset.userId);  // "42" — kebab-case → camelCase
console.log(div.dataset.role);    // "admin"
div.dataset.newProp = "value";   // adds data-new-prop

// ── Styles ────────────────────────────────────────────────────
el.style.color = "red";
el.style.backgroundColor = "#f0f";  // camelCase for CSS props
el.style.fontSize = "1.5rem";
el.style.cssText = "color: red; font-size: 1rem;"; // set multiple at once

// Read computed style (actual applied value, including CSS file rules)
const computed = window.getComputedStyle(el);
console.log(computed.color);    // "rgb(255, 0, 0)"

// ── CSS classes — preferred over .style ───────────────────────
el.classList.add("active");
el.classList.remove("hidden");
el.classList.toggle("dark");           // add if absent, remove if present
el.classList.replace("old", "new");
console.log(el.classList.contains("active"));  // true
// Prefer CSS classes over inline styles — separation of concerns
// 03
Adding & Removing Elements
Dynamically create, insert, move and delete DOM nodes.
dom_create.js 
// ── Create elements ───────────────────────────────────────────
const card = document.createElement("div");
card.className = "card";
card.dataset.id = "42";

const title = document.createElement("h2");
title.textContent = "Card Title";

const body = document.createElement("p");
body.textContent = "Card body text";

// ── Append / insert ───────────────────────────────────────────
card.append(title, body);        // append multiple children
card.prepend(document.createElement("span"));  // insert at start

document.body.append(card);     // add to page

// insertAdjacentElement — precise positioning
const ref = document.querySelector(".existing");
ref.insertAdjacentElement("beforebegin", card);  // before ref itself
ref.insertAdjacentElement("afterbegin", card);   // first child of ref
ref.insertAdjacentElement("beforeend", card);    // last child of ref
ref.insertAdjacentElement("afterend", card);     // after ref itself

// insertAdjacentHTML — fast but XSS-risky
ref.insertAdjacentHTML("beforeend", "<li>new item</li>");

// ── Remove elements ───────────────────────────────────────────
card.remove();                   // remove from DOM directly
ref.removeChild(title);          // remove a specific child

// ── Clone an element ──────────────────────────────────────────
const template = document.querySelector(".card-template");
const clone = template.cloneNode(true);  // true = deep clone (includes children)
clone.querySelector("h2").textContent = "New Title";
document.body.append(clone);

// ── DocumentFragment — batch DOM updates ─────────────────────
// Avoids multiple reflows — attach all at once
const fragment = document.createDocumentFragment();
["Alice", "Bob", "Charlie"].forEach(name => {
  const li = document.createElement("li");
  li.textContent = name;
  fragment.append(li);              // modify fragment, not live DOM
});
document.querySelector("ul").append(fragment);  // ONE DOM update

Event Listeners
Events are signals fired by the browser when things happen. You listen for them with addEventListener.
events.js 
const btn = document.querySelector("#btn");

// ── addEventListener ──────────────────────────────────────────
// Preferred — multiple listeners, explicit removal, options
btn.addEventListener("click", (event) => {
  console.log("Clicked!", event);
});

// ── The Event object ──────────────────────────────────────────
btn.addEventListener("click", (e) => {
  console.log(e.type);           // "click"
  console.log(e.target);         // element that was clicked
  console.log(e.currentTarget);  // element listener is on
  console.log(e.clientX, e.clientY);  // mouse position
  console.log(e.timeStamp);      // when it happened (ms)
});

// ── Remove event listener ─────────────────────────────────────
// Must pass the EXACT SAME function reference to remove it
function handleClick(e) { console.log("click"); }
btn.addEventListener("click", handleClick);
btn.removeEventListener("click", handleClick);  // works ✅
// btn.removeEventListener("click", () => {}) ← ❌ won't work (different ref)

// ── once option — auto-remove after first fire ────────────────
btn.addEventListener("click", handleClick, { once: true });

// ── Common event types ────────────────────────────────────────
// Mouse:    click, dblclick, mousedown, mouseup, mousemove,
//           mouseenter, mouseleave, contextmenu
// Keyboard: keydown, keyup, keypress (deprecated)
// Form:     submit, change, input, focus, blur, reset
// Window:   load, DOMContentLoaded, resize, scroll, unload
// Touch:    touchstart, touchend, touchmove

// ── Keyboard events ───────────────────────────────────────────
document.addEventListener("keydown", (e) => {
  console.log(e.key);     // "Enter", "a", "ArrowLeft", etc.
  console.log(e.code);    // "Enter", "KeyA", "ArrowLeft" (layout-independent)
  console.log(e.ctrlKey); // true if Ctrl held
  console.log(e.shiftKey);// true if Shift held

  if (e.key === "Escape") closeModal();
  if (e.ctrlKey && e.key === "s") { e.preventDefault(); save(); }
});

// ── Form events ───────────────────────────────────────────────
const form = document.querySelector("form");
form.addEventListener("submit", (e) => {
  e.preventDefault();            // stop page reload
  const data = new FormData(form);
  console.log(data.get("username"));
});

Event Bubbling & Capturing
Events travel through the DOM tree in two phases. Understanding this unlocks event delegation and prevents subtle bugs.

Three phases of event propagation

Capture phase — event travels DOWN from document → target.
Target phase — event fires on the actual target element.
Bubble phase — event travels UP from target → document.
By default, addEventListener listens in the bubble phase. Pass true or { capture: true } to listen in the capture phase.

bubbling.js 
// HTML: <div id="outer"><div id="middle"><button id="btn">Click</button></div></div>

const outer  = document.getElementById("outer");
const middle = document.getElementById("middle");
const btn    = document.getElementById("btn");

// ── Bubbling (default) ────────────────────────────────────────
btn.addEventListener("click", (e) => console.log("1. btn"));
middle.addEventListener("click", (e) => console.log("2. middle"));
outer.addEventListener("click", (e) => console.log("3. outer"));
// Clicking btn fires: 1. btn → 2. middle → 3. outer (bubbles UP)

// ── Capture phase ────────────────────────────────────────────
outer.addEventListener("click", (e) => console.log("capture: outer"), true);
// Fires BEFORE button's own handler — capture travels DOWN first

// ── Stop bubbling ─────────────────────────────────────────────
btn.addEventListener("click", (e) => {
  e.stopPropagation();  // stops bubbling — middle and outer won't fire
  console.log("btn only");
});

// stopImmediatePropagation — also stops other listeners on SAME element
btn.addEventListener("click", (e) => {
  e.stopImmediatePropagation();
  console.log("this runs");
});
btn.addEventListener("click", () => console.log("this DOESN'T run"));

// ── Prevent default browser behaviour ────────────────────────
document.querySelector("a").addEventListener("click", (e) => {
  e.preventDefault();   // stops navigation
  console.log("Link clicked but not followed");
});
document.querySelector("form").addEventListener("submit", (e) => {
  e.preventDefault();   // stops page reload
});

// ── e.target vs e.currentTarget ──────────────────────────────
outer.addEventListener("click", (e) => {
  console.log(e.target);        // element that was ACTUALLY clicked (btn)
  console.log(e.currentTarget); // element listener is ON (outer)
});
click on btn fires in order
▶capture: outer ← capture, travels DOWN
▶1. btn ← target phase
▶2. middle ← bubble, travels UP
▶3. outer ← bubble, travels UP
// 06
Event Delegation
Instead of attaching listeners to every child, attach one listener to the parent and use event.target to identify which child was interacted with.

Why delegation is powerful

Event delegation leverages bubbling. One parent listener handles events from all current and future children — you don't need to re-attach listeners when elements are added dynamically. It's also more memory-efficient than attaching one listener per element in large lists.

delegation.js 
// ── Without delegation (bad for dynamic content) ──────────────
// Attaching one listener per item — won't catch future items
document.querySelectorAll(".item").forEach(item => {
  item.addEventListener("click", () => console.log("item clicked"));
});
// ❌ Items added later won't have this listener

// ── With delegation (handles current + future items) ──────────
const list = document.querySelector("#list");

list.addEventListener("click", (e) => {
  // Check if the click was on or inside an .item
  const item = e.target.closest(".item");
  if (!item) return;  // clicked elsewhere in the list

  console.log("Item clicked:", item.dataset.id);
});
// ✅ Works for items added dynamically too

// ── Delegation with multiple actions ─────────────────────────
const taskList = document.querySelector("#tasks");

taskList.addEventListener("click", (e) => {
  const { action } = e.target.dataset;  // <button data-action="delete">
  const taskEl = e.target.closest("[data-task-id]");
  if (!action || !taskEl) return;

  const id = taskEl.dataset.taskId;

  switch (action) {
    case "delete":  deleteTask(id);  break;
    case "edit":    editTask(id);    break;
    case "complete": completeTask(id); break;
  }
});
// One listener handles all button types for all tasks — past and future

Fetch API & Network Requests
The Fetch API is the modern way to make HTTP requests from the browser. It returns Promises.
fetch_api.js 
const API = "https://jsonplaceholder.typicode.com";

// ── GET request ───────────────────────────────────────────────
async function getUsers() {
  const response = await fetch(`${API}/users`);

  // fetch only rejects on network failure — not on 404/500!
  if (!response.ok) {
    throw new Error(`HTTP ${response.status}: ${response.statusText}`);
  }

  const users = await response.json();  // parse JSON body
  return users;
}

// ── POST request ──────────────────────────────────────────────
async function createPost(postData) {
  const response = await fetch(`${API}/posts`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": "Bearer mytoken"
    },
    body: JSON.stringify(postData)   // must stringify for JSON
  });

  if (!response.ok) throw new Error(`Error: ${response.status}`);
  return response.json();
}

// ── DELETE / PUT / PATCH ──────────────────────────────────────
async function deletePost(id) {
  const res = await fetch(`${API}/posts/${id}`, { method: "DELETE" });
  return res.ok;
}

// ── Response object methods ───────────────────────────────────
// response.json()    — parse as JSON (most common)
// response.text()    — parse as plain text
// response.blob()    — parse as Blob (images, files)
// response.formData()— parse as FormData
// response.ok        — true if status 200–299
// response.status    — e.g. 200, 404, 500
// response.headers   — Headers object

// ── CORS — Cross-Origin Requests ─────────────────────────────
// Browsers block requests to different origins by default
// CORS headers on the server allow specific origins:
//   Access-Control-Allow-Origin: https://mysite.com
// Preflight OPTIONS request is sent for complex requests first
// Can't bypass CORS from the browser — must be allowed server-side

// ── Abort a fetch (timeout pattern) ─────────────────────────
async function fetchWithTimeout(url, ms = 5000) {
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), ms);

  try {
    const res = await fetch(url, { signal: controller.signal });
    clearTimeout(timeoutId);
    return res.json();
  } catch (e) {
    if (e.name === "AbortError") throw new Error("Request timed out");
    throw e;
  }
}
